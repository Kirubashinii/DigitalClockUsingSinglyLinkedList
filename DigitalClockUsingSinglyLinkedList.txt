#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <unistd.h>

// Define Node structure
struct Node {
    int timeUnit;
    struct Node* next;
};

// Function to create a new node
struct Node* createNode() {
    struct Node* newNode = (struct Node*) malloc(sizeof(struct Node));
    newNode->timeUnit = 0;
    newNode->next = NULL;
    return newNode;
}

// Function to create a clock list: hour -> minute -> second
struct Node* createClockList() {
    struct Node* hour = createNode();
    struct Node* minute = createNode();
    struct Node* second = createNode();

    hour->next = minute;
    minute->next = second;

    return hour;  // Head pointer
}

// Function to update time in the list from system clock
void updateClockList(struct Node* head) {
    time_t now = time(NULL);
    struct tm* t = localtime(&now);

    head->timeUnit = t->tm_hour;
    head->next->timeUnit = t->tm_min;
    head->next->next->timeUnit = t->tm_sec;
}

// Function to display the current time
void displayClock(struct Node* head) {
    printf("\r%02d:%02d:%02d", 
           head->timeUnit, 
           head->next->timeUnit, 
           head->next->next->timeUnit);
    fflush(stdout);
}

int main() {
    struct Node* clockHead = createClockList();

    printf("Digital Clock (Singly Linked List Version 2):\n");

    while (1) {
        updateClockList(clockHead);
        displayClock(clockHead);
        sleep(1);
    }

    return 0;
}
